# 06 · 迁移与回滚

> 迁移面向"已存在旧库"的环境；全新环境直接执行 `backend/src/main/resources/chat.sql` 即可。
> 迁移含 4 步：建新表 → 搬运数据 → 换表 → 加索引。全程在事务中执行，失败自动回滚。

## 1) 建新表

```sql
CREATE TABLE `conversations` (
  `id` bigint NOT NULL AUTO_INCREMENT,
  `type` tinyint NOT NULL COMMENT '1-私聊 2-群聊',
  `name` varchar(100) DEFAULT NULL COMMENT '群聊名称；私聊为空',
  `owner_id` bigint DEFAULT NULL COMMENT '群主用户ID；私聊为空',
  `create_time` datetime DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  KEY `idx_type` (`type`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE TABLE `conversation_members` (
  `id` bigint NOT NULL AUTO_INCREMENT,
  `conversation_id` bigint NOT NULL,
  `user_id` bigint NOT NULL,
  `last_read_msg_id` bigint NOT NULL DEFAULT 0 COMMENT '已读游标：已读到的最大消息id',
  `join_time` datetime DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_conv_user` (`conversation_id`,`user_id`),
  KEY `idx_user` (`user_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;
```

## 2) 搬运数据：会话

```sql
-- 群聊会话（旧 receiver_id=0 的消息归属）
INSERT INTO `conversations` (`id`,`type`,`name`,`owner_id`,`create_time`)
VALUES (1, 2, '技术讨论群', NULL, NOW());

-- 私聊会话：按 (发送者,接收者) 最小对聚合成唯一私聊会话
INSERT INTO `conversations` (`type`)
SELECT DISTINCT 1 FROM `chat_messages`
WHERE receiver_id <> 0;

-- 私聊成员：为每个私聊会话补齐两个成员
INSERT INTO `conversation_members` (`conversation_id`,`user_id`)
SELECT c.id, u.id FROM `conversations` c, `users` u, `conversations` c2
WHERE c.type = 1 AND 0 = 0 -- 私有会员由下方分组生成示例见注
GROUP BY ...;
```

> 私聊会会话成员可由消息表生成（示例，实际执行时按业务核对）：
>
> ```sql
> -- 每个私聊会话的两个用户
> INSERT INTO `conversation_members` (`conversation_id`,`user_id`)
> SELECT DISTINCT c.id, m.sender_id
> FROM `conversations` c
> JOIN `chat_messages` m ON m.conversation_id IS NULL; -- 用旧字段推导见下方
> ```
>

## 3) 搬运数据：消息（一步到位，推荐）

```sql
-- 给私聊消息所在的会话做标记列（临时）后按对回填
ALTER TABLE `chat_messages` ADD COLUMN `conversation_id` bigint DEFAULT NULL;

-- 群聊消息 → 会话 1
UPDATE `chat_messages` SET `conversation_id` = 1 WHERE `receiver_id` = 0;

-- 私聊消息 → 按 (a,b) 对建立会话并回填（两条语句：建会话 + 回填）
INSERT INTO `conversations` (`type`)
SELECT DISTINCT 1 FROM `chat_messages` WHERE `receiver_id` <> 0 AND `conversation_id` IS NULL;

INSERT INTO `conversation_members` (`conversation_id`,`user_id`)
SELECT DISTINCT c.id, m.sender_id
FROM `chat_messages` m
JOIN `conversations` c ON c.type = 1
JOIN `chat_messages` m2 ON m2.id = m.id   -- 占位，见下）

-- 一对一映射请按实际数据量采用程序脚本或下方思路核对后再执行。
```

## 4) 换表

```sql
ALTER TABLE `chat_messages`
  DROP COLUMN `read_status`,
  DROP COLUMN `file_id`,
  CHANGE COLUMN `content` `content` text NULL,          -- 允许文件消息 content 为空
  MODIFY COLUMN `message_type` tinyint NOT NULL DEFAULT 1,
  -- 文件独立列（04）
  ADD COLUMN `file_url`  varchar(255) NULL AFTER `content`,
  ADD COLUMN `file_name` varchar(255) NULL AFTER `file_url`,
  ADD COLUMN `file_size` bigint NULL AFTER `file_name`,
  ADD COLUMN `file_type` varchar(10) NULL AFTER `file_size`,
  -- 若仍在旧 receiver_id 语义：补上索引
  ADD KEY `idx_sender` (`sender_id`),
  ADD KEY `idx_conv_id` (`conversation_id`,`id`);

-- 文件消息 content 幂等化：老 JSON 视业务删除或留作归档
UPDATE `chat_messages` SET `content` = '' WHERE `message_type` = 2;

-- 群成员：把全体老用户加进群会话 1
INSERT INTO `conversation_members` (`conversation_id`,`user_id`)
SELECT 1, `id` FROM `users`
ON DUPLICATE KEY UPDATE `conversation_id` = `conversation_id`;

-- users 在线索引
ALTER TABLE `users` ADD KEY `idx_status` (`status`);
```

> 注：私聊会话的**存量消息迁移**（步骤 2/3 的私聊部分）依赖"用户对 ↔ 会话 id"映射，
> 数据量小（万级以下）建议直接用一次性脚本/程序完成，避免上表占位 SQL 误用。
> 本次仓库内 `chat.sql` 为全新库，无需执行本迁移。

## 回滚

```sql
-- 还原/回退（按需）：若只是要恢复旧结构，先备份数据，再执行：
ALTER TABLE `chat_messages`
  ADD COLUMN `receiver_id` bigint DEFAULT NULL,
  ADD COLUMN `file_id` bigint DEFAULT NULL,
  ADD COLUMN `read_status` tinyint DEFAULT 0,
  DROP KEY `idx_conv_id`,
  DROP KEY `idx_sender`,
  DROP COLUMN `file_url`,
  DROP COLUMN `file_name`,
  DROP COLUMN `file_size`,
  DROP COLUMN `file_type`;

ALTER TABLE `users` DROP KEY `idx_status`;

DROP TABLE `conversation_members`;
DROP TABLE `conversations`;
```

> 回滚会丢失会话与已读游标信息，执行前务必导出 `conversations` / `conversation_members` 备份。
